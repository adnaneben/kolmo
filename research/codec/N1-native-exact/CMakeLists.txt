cmake_minimum_required(VERSION 3.20.2)
project(n1_native_exact LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT OPENZL_SOURCE)
 message(FATAL_ERROR "Pass -DOPENZL_SOURCE=... to verified OpenZL 32246b48 checkout")
endif()
foreach(opt BUILD_ALL BUILD_TESTS BUILD_BENCHMARKS BUILD_PARQUET_TOOLS BUILD_PYTHON_EXT BUILD_PYTHON_DEMO BUILD_CPP BUILD_CUSTOM_PARSERS BUILD_TOOLS BUILD_CLI BUILD_EXAMPLES INSTALL CPP_INSTALL ALLOW_INTROSPECTION)
 set(OPENZL_${opt} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(${OPENZL_SOURCE} openzl-build EXCLUDE_FROM_ALL)
add_executable(native_exact adapter.cpp)
target_include_directories(native_exact PRIVATE ${OPENZL_SOURCE})
target_link_libraries(native_exact PRIVATE openzl)
add_executable(parse_cli parse_cli.cpp)
if(ADAPTER_SANITIZE)
 foreach(t native_exact parse_cli)
  target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${t} PRIVATE -fsanitize=address,undefined)
 endforeach()
endif()
add_executable(stock_reader stock_reader.c)
target_link_libraries(stock_reader PRIVATE openzl)
add_executable(native_exact_san adapter.cpp)
target_include_directories(native_exact_san PRIVATE ${OPENZL_SOURCE})
target_link_libraries(native_exact_san PRIVATE openzl)
target_compile_options(native_exact_san PRIVATE -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(native_exact_san PRIVATE -fsanitize=address,undefined)
